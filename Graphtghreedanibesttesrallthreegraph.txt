 *  Features: 3D Rotating Specimen, Stress Heatmap, Necking, SSE

// STATES
final int MEASURING  = -1;
final int IDLE       = 0;
final int LOADING    = 1;
final int UNLOADING  = 2;
final int RELOADING  = 3;
final int FRACTURED  = 4;
int state = MEASURING;

// MEASURING PHASE VARIABLES
int measStep = 0;
float calX = 650, calY = 80, calRot = 0, calJaw = 250;
float targetX = 650, targetY = 80, targetRot = 0, targetJaw = 250;
Button btnMeasure;

// PHYSICS / MATERIAL MODEL (Stress-Strain Curve)
float strain = 0;
float stress = 0;
float maxStrainReached = 0; 
float unloadTargetStrain = 0;

final float ELASTIC_LIMIT   = 100.0f;
final float ELASTIC_SLOPE   = 2.5f;
final float YIELD_STRESS    = 250.0f;
final float UTS_STRAIN      = 260.0f;
final float FRACTURE_STRAIN = 360.0f;
final float PEAK_STRESS     = 330.0f;

final float LOAD_SPEED   = 0.8f;
final float UNLOAD_SPEED = 1.6f;

// GRAPH TRACE MEMORY
class TracePoint {
  float e, s;
  TracePoint(float e, float s) {
    this.e = e; this.s = s;
  }
}
ArrayList<TracePoint> tracePath = new ArrayList<TracePoint>();

// GRAPH LAYOUT
final float LOGICAL_WIDTH  = 900;
final float LOGICAL_HEIGHT = 600;
final float graphX      = 80;
final float graphY      = 495;
final float graphWidth  = 430;
final float graphHeight = 400;

// BUTTONS
Button btnStart, btnInterrupt, btnReset;

// SCREEN SCALING
float scaleFactor = 1.0f;
float offsetX = 0, offsetY = 0;

// ANIMATION VARIABLES
float[] cachedStress;
float fractureFlash = 0;
float specimenShake = 0;
float timeSeconds = 0;

void setup() {
  // P3D enables 3D rendering for the cylindrical bar
  fullScreen(P3D);
  orientation(LANDSCAPE);
  frameRate(60);
  smooth(4);

  // Pre-calculate the Stress-Strain Curve (SSE) geometry
  cachedStress = new float[(int)FRACTURE_STRAIN + 3];
  for (int i = 0; i <= FRACTURE_STRAIN + 2; i++) {
    cachedStress[i] = calculateTheoreticalStress((float)i);
  }

  tracePath.add(new TracePoint(0, 0));

  // Init Buttons
  btnMeasure   = new Button("MEASURE d1 (Top)", 50, 520, 250, 45, color(31, 111, 235));
  btnStart     = new Button("APPLY LOAD", 150, 540, 150, 45, color(35, 134, 54));
  btnInterrupt = new Button("UNLOAD (Interrupt)", 320, 540, 160, 45, color(215, 58, 73));
  btnReset     = new Button("RESET LAB", 500, 540, 120, 45, color(45, 50, 58));

  updateScreenScale();
}

void draw() {
  background(13, 17, 23);
  updateScreenScale();
  
  timeSeconds += 1.0f / 60.0f;

  pushMatrix();
  translate(offsetX, offsetY);
  scale(scaleFactor);

  if (state == MEASURING) {
    drawMeasurementPhase();
  } else {
    drawTensileTestPhase();
  }

  popMatrix();
}

void updateScreenScale() {
  scaleFactor = min((float)width / LOGICAL_WIDTH, (float)height / LOGICAL_HEIGHT);
  if (scaleFactor < 0.01f) scaleFactor = 1.0f;
  offsetX = (width - LOGICAL_WIDTH * scaleFactor) / 2.0f;
  offsetY = (height - LOGICAL_HEIGHT * scaleFactor) / 2.0f;
}

// ================================================================
// PHASE 1: MEASUREMENT (VERNIER CALIPER)
// ================================================================
void drawMeasurementPhase() {
  calX = lerp(calX, targetX, 0.1f);
  calY = lerp(calY, targetY, 0.1f);
  calRot = lerp(calRot, targetRot, 0.1f);
  calJaw = lerp(calJaw, targetJaw, 0.1f);

  hint(DISABLE_DEPTH_TEST);
  fill(227, 179, 65); textAlign(LEFT, TOP); textSize(24);
  text("PHASE 1: INITIAL MEASUREMENTS", 50, 30);
  fill(180); textSize(14);
  text("Measure the 3D bar diameter at 3 points, then the gauge length.", 50, 65);
  
  drawNotebook();
  hint(ENABLE_DEPTH_TEST);

  // Draw static 3D specimen for measurement
  draw3DSpecimen(450, 300, 0, 0, 0, false);

  hint(DISABLE_DEPTH_TEST);
  drawVernierCaliper(calX, calY, calRot, calJaw);
  btnMeasure.display();
  hint(ENABLE_DEPTH_TEST);
}

// ================================================================
// PHASE 2: 3D TENSILE TEST & PHYSICS
// ================================================================
void drawTensileTestPhase() {
  if (fractureFlash > 0) fractureFlash -= 4.0f;
  if (fractureFlash < 0) fractureFlash = 0;

  updatePhysics();

  // 1. Draw 2D UI Background layers
  hint(DISABLE_DEPTH_TEST);
  drawGraphPanel();
  drawTheoreticalCurve();
  drawActiveCurve();
  drawGraphMarkers();
  hint(ENABLE_DEPTH_TEST);

  // 2. Draw 3D Rotating Heatmap Specimen
  lights();
  directionalLight(200, 200, 200, 0.5f, 0, -1);
  drawActive3DSpecimen();
  noLights();

  // 3. Draw 2D UI Foregrounds
  hint(DISABLE_DEPTH_TEST);
  drawInstructor();
  drawStatusPanel();
  
  btnStart.display();
  btnInterrupt.display();
  btnReset.display();
  
  // Flash effect on fracture
  if (fractureFlash > 0) {
    fill(255, 255, 255, fractureFlash);
    noStroke();
    rect(0, 0, LOGICAL_WIDTH, LOGICAL_HEIGHT);
  }
  hint(ENABLE_DEPTH_TEST);
}

// --- Physics Logic ---
float calculateTheoreticalStress(float e) {
  if (e <= 0) return 0;
  if (e <= ELASTIC_LIMIT) return e * ELASTIC_SLOPE;
  if (e <= UTS_STRAIN) {
    float x = (e - ELASTIC_LIMIT) / (UTS_STRAIN - ELASTIC_LIMIT);
    return YIELD_STRESS + (PEAK_STRESS - YIELD_STRESS) * x * x;
  }
  if (e <= FRACTURE_STRAIN) {
    float x = (e - UTS_STRAIN) / (FRACTURE_STRAIN - UTS_STRAIN);
    return PEAK_STRESS - 55.0f * pow(x, 1.25f);
  }
  return 0;
}

float getFastStress(float e) {
  if (e <= 0) return 0;
  if (e >= FRACTURE_STRAIN) return cachedStress[(int)FRACTURE_STRAIN];
  int lower = floor(e);
  float fraction = e - lower;
  return lerp(cachedStress[lower], cachedStress[lower + 1], fraction);
}

void updatePhysics() {
  if (state == LOADING) {
    strain += LOAD_SPEED;
    if (strain >= FRACTURE_STRAIN) {
      strain = FRACTURE_STRAIN;
      stress = getFastStress(strain);
      state = FRACTURED;
      tracePath.add(new TracePoint(strain, stress));
      fractureFlash = 255; specimenShake = 15;
      return;
    }
    stress = getFastStress(strain);
    maxStrainReached = max(maxStrainReached, strain);

    if (strain > UTS_STRAIN) {
      specimenShake = sin(timeSeconds * 20.0f) * ((strain - UTS_STRAIN) / 100.0f);
    }
  } else if (state == UNLOADING) {
    strain -= UNLOAD_SPEED;
    stress -= UNLOAD_SPEED * ELASTIC_SLOPE;
    if (stress <= 0 || strain <= unloadTargetStrain) {
      stress = 0;
      strain = max(0, unloadTargetStrain);
      state = IDLE;
      tracePath.add(new TracePoint(strain, stress));
    }
  } else if (state == RELOADING) {
    strain += LOAD_SPEED;
    stress += LOAD_SPEED * ELASTIC_SLOPE;
    if (strain >= maxStrainReached) {
      strain = maxStrainReached;
      stress = getFastStress(strain);
      state = LOADING;
      tracePath.add(new TracePoint(strain, stress));
    }
  }

  if (specimenShake != 0) {
    specimenShake *= 0.90f;
    if (abs(specimenShake) < 0.05f) specimenShake = 0;
  }

  if (state != IDLE && state != FRACTURED) {
    TracePoint lastPt = tracePath.get(tracePath.size() - 1);
    if (lastPt == null || dist(lastPt.e, lastPt.s, strain, stress) > 0.5f) {
      tracePath.add(new TracePoint(strain, stress));
    }
  }
}

// ================================================================
// 3D SPECIMEN & HEATMAP RENDERER
// ================================================================
void drawActive3DSpecimen() {
  float specX = 720, specY = 280;
  float shakeX = (state == FRACTURED) ? sin(timeSeconds * 30) * specimenShake : 0;
  draw3DSpecimen(specX + shakeX, specY, strain, stress, maxStrainReached, state == FRACTURED);
}

void draw3DSpecimen(float x, float y, float simStrain, float simStress, float maxStr, boolean isFractured) {
  float baseLength = 160;
  float baseRadius = 24;
  
  // Physical scaling based on strain
  float currentLength = baseLength + simStrain * 0.43f;
  float physStrain = simStrain / 1000.0f;
  
  // Poisson's ratio (elastic -> plastic volume conservation)
  float nu = (simStrain > ELASTIC_LIMIT) ? lerp(0.3f, 0.45f, (simStrain-ELASTIC_LIMIT)/100.0f) : 0.3f;
  float uniformRadius = baseRadius * (1.0f - nu * physStrain);

  // Localized Necking calculation
  float neckMax = 0;
  if (maxStr > UTS_STRAIN) {
    neckMax = (maxStr - UTS_STRAIN) * 0.22f;
    neckMax = constrain(neckMax, 0, uniformRadius * 0.70f); 
  }

  pushMatrix();
  translate(x, y, 0);
  rotateX(-0.1f); // Tilt slightly towards viewer
  rotateY(frameCount * 0.02f); // Continuous spin

  // Grips
  noStroke();
  fill(40, 46, 54);
  drawCylinder(0, -currentLength/2 - 20, baseRadius + 15, baseRadius + 15, 40);
  drawCylinder(0, currentLength/2 + 20, baseRadius + 15, baseRadius + 15, 40);

  // Deformation Mesh (Gauge Length)
  int stacks = 35;
  int slices = 24;
  float halfL = currentLength / 2.0f;
  float gap = isFractured ? 8 : 0;

  for (int i = 0; i < stacks; i++) {
    float y1 = map(i, 0, stacks, -halfL, halfL);
    float y2 = map(i+1, 0, stacks, -halfL, halfL);

    // Skip drawing the exact middle if fractured
    if (isFractured && y1 > -gap && y2 < gap) continue;

    beginShape(QUAD_STRIP);
    for (int j = 0; j <= slices; j++) {
      float theta = map(j, 0, slices, 0, TWO_PI);

      // Apply Gaussian Necking curve
      float n1 = neckMax * exp(-pow(y1, 2) / 600.0f);
      float r1 = uniformRadius - n1;
      
      float n2 = neckMax * exp(-pow(y2, 2) / 600.0f);
      float r2 = uniformRadius - n2;

      // Calculate True Stress mapping for heatmap
      float trueStress1 = calculateTrueStress(simStress, baseRadius, r1);
      float trueStress2 = calculateTrueStress(simStress, baseRadius, r2);

      fill(getStressColor(trueStress1));
      vertex(r1 * cos(theta), y1, r1 * sin(theta));
      
      fill(getStressColor(trueStress2));
      vertex(r2 * cos(theta), y2, r2 * sin(theta));
    }
    endShape();
  }
  popMatrix();
}

float calculateTrueStress(float nominal, float r0, float rCurrent) {
  if (nominal <= 0) return 0;
  // True Stress = Nominal Stress * (Original Area / Current Area)
  return nominal * ((r0 * r0) / (rCurrent * rCurrent));
}

int getStressColor(float localStress) {
  int steelColor = color(190, 200, 210);
  if (localStress <= 1) return steelColor;
  
  // Heatmap boundaries
  float maxTrueStress = PEAK_STRESS * 3.5f; 
  float t = constrain(localStress / maxTrueStress, 0, 1);
  
  if (t < 0.25f) return lerpColor(steelColor, color(255, 230, 0), t / 0.25f);
  if (t < 0.50f) return lerpColor(color(255, 230, 0), color(255, 100, 0), (t - 0.25f) / 0.25f);
  if (t < 0.75f) return lerpColor(color(255, 100, 0), color(255, 20, 20), (t - 0.50f) / 0.25f);
  return lerpColor(color(255, 20, 20), color(255, 255, 255), (t - 0.75f) / 0.25f);
}

void drawCylinder(float x, float y, float rTop, float rBot, float h) {
  pushMatrix();
  translate(x, y, 0);
  beginShape(QUAD_STRIP);
  for(int j=0; j<=24; j++) {
    float theta = map(j, 0, 24, 0, TWO_PI);
    vertex(rTop * cos(theta), -h/2, rTop * sin(theta));
    vertex(rBot * cos(theta), h/2, rBot * sin(theta));
  }
  endShape();
  popMatrix();
}

void drawInstructor() {
  float x = 845, y = 495;
  String l1 = "Press Apply Load", l2 = "Watch the graph & 3D bar";

  if (state == LOADING) {
    if (strain < ELASTIC_LIMIT) { l1 = "Elastic Deformation"; l2 = "Bonds stretching. Reversible."; }
    else if (strain < UTS_STRAIN) { l1 = "Plastic Deformation"; l2 = "Dislocations moving. Permanent set."; }
    else { l1 = "Necking Region"; l2 = "Area dropping. True stress spikes!"; }
  } else if (state == UNLOADING) { l1 = "Unloading phase"; l2 = (unloadTargetStrain < 2) ? "Full elastic recovery" : "Permanent plastic set left!"; }
  else if (state == FRACTURED) { l1 = "FRACTURE!"; l2 = "Atomic bonds broken."; }

  noStroke(); fill(255); rect(x - 170, y - 115, 170, 58, 10);
  triangle(x - 70, y - 57, x - 50, y - 57, x - 40, y - 35);
  fill(20); textAlign(CENTER, CENTER); textSize(12);
  text(l1, x - 85, y - 98); text(l2, x - 85, y - 78);
}

void drawStatusPanel() {
  float x = 555, y = 30;
  fill(23, 28, 35, 230); noStroke(); rect(x, y, 320, 160, 10);

  String stateStr = "IDLE";
  int stColor = color(255);
  if (state == LOADING) { stateStr = "LOADING"; stColor = color(70,220,110); }
  else if (state == UNLOADING) { stateStr = "UNLOADING"; stColor = color(255,100,100); }
  else if (state == RELOADING) { stateStr = "RELOADING"; stColor = color(220,220,70); }
  else if (state == FRACTURED) { stateStr = "FRACTURED"; stColor = color(255,60,60); }
  
  fill(180); textAlign(LEFT, CENTER); textSize(12); text("MATERIAL STATE", x + 15, y + 20);
  fill(stColor); textSize(16); text(stateStr, x + 130, y + 20);

  float physStrain = strain / 1000.0f;
  float initL = 50.00f, initD = 12.50f;
  float deltaL = physStrain * initL;
  float nu = (strain > ELASTIC_LIMIT) ? lerp(0.3f, 0.45f, (strain-ELASTIC_LIMIT)/100.0f) : 0.3f;
  
  // Calculate average D including necking reduction for display
  float neck = (maxStrainReached > UTS_STRAIN) ? (maxStrainReached-UTS_STRAIN)*0.035f : 0;
  float deltaD = -nu * physStrain * initD - neck;

  fill(180); textSize(12);
  text("Nom. Stress:", x + 15, y + 55); fill(255); textSize(14); text(nf(stress, 0, 1) + " MPa", x + 95, y + 55);
  fill(180); textSize(12);
  text("Strain:", x + 185, y + 55); fill(255); textSize(14); text(nf(physStrain, 0, 3), x + 235, y + 55);

  fill(150, 200, 255); textSize(13);
  text("L0:  50.00 mm", x + 15, y + 85);
  text("D0:  12.50 mm", x + 175, y + 85);

  fill(255, 170, 170);
  text("ΔL: +" + nf(abs(deltaL), 0, 2) + " mm", x + 15, y + 110);
  text("Δd: -" + nf(abs(deltaD), 0, 3) + " mm", x + 175, y + 110);

  fill(150, 255, 150);
  text("L:  " + nf(initL + deltaL, 0, 2) + " mm", x + 15, y + 135);
  text("d:  " + nf(initD + deltaD, 0, 2) + " mm", x + 175, y + 135);
}

// Graph and line drawing helpers
void drawGraphPanel() {
  stroke(220); strokeWeight(2); noFill();
  line(graphX, graphY, graphX, graphY - graphHeight);
  line(graphX, graphY, graphX + graphWidth, graphY);

  stroke(255, 18); strokeWeight(1);
  for (int y = 0; y <= 350; y += 50) line(graphX, graphY - y, graphX + graphWidth, graphY - y);
  for (int x = 0; x <= 400; x += 50) line(graphX + x, graphY, graphX + x, graphY - graphHeight);

  fill(255); noStroke(); textAlign(RIGHT, CENTER); textSize(16);
  text("STRESS", graphX - 15, graphY - graphHeight + 10);
  textAlign(CENTER, TOP); text("STRAIN", graphX + graphWidth / 2, graphY + 15);

  fill(227, 179, 65); textAlign(LEFT, CENTER); textSize(18);
  text("PHASE 2: SSE & DEFORMATION", graphX + 80, graphY - graphHeight + 25);
}

void drawTheoreticalCurve() {
  noFill(); stroke(255, 35); strokeWeight(2);
  beginShape();
  for (int e = 0; e <= FRACTURE_STRAIN; e += 2) vertex(graphX + e, graphY - cachedStress[e]);
  endShape();
}

void drawActiveCurve() {
  if (tracePath.size() == 0) return;
  noFill(); stroke(0, 255, 255); strokeWeight(3);
  beginShape();
  for (TracePoint pt : tracePath) vertex(graphX + pt.e, graphY - pt.s);
  vertex(graphX + strain, graphY - stress);
  endShape();

  noStroke(); fill(255);
  ellipse(graphX + strain, graphY - stress, 11, 11);
}

void drawGraphMarkers() {
  drawMarker(ELASTIC_LIMIT, "Yield", 100, color(100, 200, 255));
  drawMarker(UTS_STRAIN, "UTS", PEAK_STRESS, color(255, 180, 80));
  drawMarker(FRACTURE_STRAIN, "Fracture", getFastStress(FRACTURE_STRAIN), color(255, 80, 80));
}

void drawMarker(float e, String label, float s, int c) {
  if (e > graphWidth) return;
  stroke(red(c), green(c), blue(c), 130); strokeWeight(1);
  dashedLine(graphX + e, graphY, graphX + e, graphY - min(s, graphHeight - 10), 5);
  noStroke(); fill(c); ellipse(graphX + e, graphY - s, 7, 7);
  textAlign(CENTER, TOP); textSize(10); text(label, graphX + e, graphY - 40);
}

// Vernier & Notebook (Phase 1)
void drawVernierCaliper(float x, float y, float rot, float jawDist) {
  pushMatrix();
  translate(x, y); rotate(rot);
  fill(220); stroke(100); strokeWeight(2);
  rect(-180, -20, 500, 40, 4);
  rect(-40, 20, 40, 120, 2);
  fill(0); noStroke();
  for(int i=0; i<=300; i+=10) {
      rect(i, -20, 2, 15);
      if(i%50==0) { textSize(10); textAlign(CENTER, TOP); text(i/10, i, -5); }
  }
  pushMatrix();
  translate(jawDist, 0);
  fill(235); stroke(100); strokeWeight(2);
  rect(-20, -25, 120, 50, 4);
  rect(0, 20, 40, 120, 2);
  fill(255, 0, 0); noStroke();
  for(int i=0; i<10; i++) rect(i*9, 15, 1.5f, 10);
  popMatrix();
  popMatrix();
}

void drawNotebook() {
  fill(250); stroke(200); strokeWeight(2);
  rect(50, 110, 250, 390, 5); 
  fill(30, 40, 50); noStroke();
  textSize(18); textAlign(CENTER, TOP); text("INSPECTION SHEET", 175, 130);
  textSize(15); textAlign(LEFT, TOP);
  float startY = 180, lh = 35; 
  if(measStep > 0) text("d1 (Top):    12.48 mm", 70, startY);
  if(measStep > 1) text("d2 (Mid):    12.52 mm", 70, startY + lh);
  if(measStep > 2) text("d3 (Bot):    12.50 mm", 70, startY + lh*2);
  if(measStep > 3) {
      stroke(100); strokeWeight(1); line(70, startY + lh*3 - 5, 230, startY + lh*3 - 5);
      noStroke(); fill(40, 130, 60); text("Avg D0:      12.50 mm", 70, startY + lh*3);
  }
  if(measStep > 4) { fill(215, 58, 73); text("Gauge L0:  50.00 mm", 70, startY + lh*5); }
}

void mousePressed() {
  if (scaleFactor <= 0.001f) return;
  float logicalX = (mouseX - offsetX) / scaleFactor;
  float logicalY = (mouseY - offsetY) / scaleFactor;

  if (state == MEASURING) {
    if (btnMeasure.isHit(logicalX, logicalY)) {
      measStep++;
      if (measStep == 1) { 
        targetX = 400; targetY = 300; targetRot = 0; targetJaw = 100;
        btnMeasure.label = "MEASURE d2 (Mid)";
      } else if (measStep == 2) { 
        targetY = 420; btnMeasure.label = "MEASURE d3 (Bot)";
      } else if (measStep == 3) { 
        targetX = 490; targetY = 150; targetRot = PI/2; targetJaw = 300;
        btnMeasure.label = "MEASURE Gauge L0";
      } else if (measStep == 4) { 
        targetX = 650; targetY = 150; targetRot = PI/2; targetJaw = 0;
        btnMeasure.label = "START TENSILE TEST";
      } else if (measStep == 5) {
        state = IDLE; 
      }
    }
    return;
  }

  if (btnStart.isHit(logicalX, logicalY)) {
    if (state == IDLE || state == UNLOADING) {
      if (strain < maxStrainReached - 0.1f) {
        state = RELOADING;
        unloadTargetStrain = strain - stress / ELASTIC_SLOPE;
      } else state = LOADING;
    }
  }

  if (btnInterrupt.isHit(logicalX, logicalY)) {
    if ((state == LOADING || state == RELOADING) && strain > 0) {
      state = UNLOADING;
      unloadTargetStrain = strain - stress / ELASTIC_SLOPE;
    }
  }

  if (btnReset.isHit(logicalX, logicalY)) resetSimulation();
}

void resetSimulation() {
  state = MEASURING; measStep = 0;
  strain = 0; stress = 0; maxStrainReached = 0; unloadTargetStrain = 0;
  tracePath.clear(); tracePath.add(new TracePoint(0, 0));
  fractureFlash = 0; specimenShake = 0;
  targetX = 650; targetY = 80; targetRot = 0; targetJaw = 250;
  calX = 650; calY = 80; calRot = 0; calJaw = 250;
  btnMeasure.label = "MEASURE d1 (Top)";
}

class Button {
  String label; float x, y, w, h; int buttonColor;
  Button(String label, float x, float y, float w, float h, int c) {
    this.label = label; this.x = x; this.y = y; this.w = w; this.h = h; this.buttonColor = c;
  }
  void display() {
    noStroke(); fill(0, 100); rect(this.x + 2, this.y + 3, this.w, this.h, 7);
    fill(this.buttonColor); rect(this.x, this.y, this.w, this.h, 7);
    fill(255, 30); rect(this.x, this.y, this.w, 15, 7);
    fill(255); textAlign(CENTER, CENTER); textSize(14);
    text(this.label, this.x + this.w / 2, this.y + this.h / 2);
  }
  boolean isHit(float mx, float my) { 
    return mx >= this.x && mx <= this.x + this.w && my >= this.y && my <= this.y + this.h; 
  }
}

void dashedLine(float x1, float y1, float x2, float y2, float dashLength) {
  float distance = dist(x1, y1, x2, y2);
  if (distance <= 0 || dashLength <= 0) return;
  int segments = max(1, ceil(distance / dashLength));
  for (int i = 0; i < segments; i += 2) {
    float sx = lerp(x1, x2, (float)i / segments);
    float sy = lerp(y1, y2, (float)i / segments);
    float ex = lerp(x1, x2, min((float)(i + 1) / segments, 1.0f));
    float ey = lerp(y1, y2, min((float)(i + 1) / segments, 1.0f));
    line(sx, sy, ex, ey);
  }
}
