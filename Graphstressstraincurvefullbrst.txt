// STATES
final int MEASURING  = -1;
final int IDLE       = 0;
final int LOADING    = 1;
final int UNLOADING  = 2;
final int RELOADING  = 3;
final int FRACTURED  = 4;
int state = MEASURING;

// MEASURING PHASE VARIABLES
int measStep = 0;
float calX = 650, calY = 80, calRot = 0, calJaw = 250;
float targetX = 650, targetY = 80, targetRot = 0, targetJaw = 250;
Button btnMeasure;

// PHYSICS / MATERIAL MODEL
float strain = 0;
float stress = 0;
float maxStrainReached = 0; 
float unloadTargetStrain = 0;

final float ELASTIC_LIMIT   = 100.0f;
final float ELASTIC_SLOPE   = 2.5f;
final float YIELD_STRESS    = 250.0f;
final float UTS_STRAIN      = 260.0f;
final float FRACTURE_STRAIN = 360.0f;
final float PEAK_STRESS     = 330.0f;

final float LOAD_SPEED   = 0.8f;
final float UNLOAD_SPEED = 1.6f;

// GRAPH TRACE MEMORY (Simulates an ink pen)
class TracePoint {
  float e, s;
  TracePoint(float e, float s) {
    this.e = e;
    this.s = s;
  }
}
ArrayList<TracePoint> tracePath = new ArrayList<TracePoint>();

// GRAPH LAYOUT
final float LOGICAL_WIDTH  = 900;
final float LOGICAL_HEIGHT = 600;
final float graphX      = 80;
final float graphY      = 495;
final float graphWidth  = 430;
final float graphHeight = 400;

// BUTTONS
Button btnStart, btnInterrupt, btnReset;

// SCREEN SCALING
float scaleFactor = 1.0f;
float offsetX = 0;
float offsetY = 0;

// GRAPH CACHE & ANIMATION
float[] cachedStress;
float fractureFlash = 0;
float specimenShake = 0;
float arrowPulse = 0;
float timeSeconds = 0;

void setup() {
  fullScreen();
  orientation(LANDSCAPE);
  frameRate(60);
  smooth();

  // Build theoretical stress curve cache
  cachedStress = new float[(int)FRACTURE_STRAIN + 3];
  for (int i = 0; i <= FRACTURE_STRAIN + 2; i++) {
    cachedStress[i] = calculateTheoreticalStress((float)i);
  }

  // Initialize trace path
  tracePath.add(new TracePoint(0, 0));

  // Setup Buttons
  btnMeasure   = new Button("MEASURE d1 (Top)", 50, 520, 250, 45, color(31, 111, 235));
  btnStart     = new Button("APPLY LOAD", 150, 540, 150, 45, color(35, 134, 54));
  btnInterrupt = new Button("UNLOAD (Interrupt)", 320, 540, 160, 45, color(215, 58, 73));
  btnReset     = new Button("RESET LAB", 500, 540, 120, 45, color(45, 50, 58));

  updateScreenScale();
}

void draw() {
  background(13, 17, 23);
  updateScreenScale();

  // Animation clock
  timeSeconds += 1.0f / 60.0f;
  arrowPulse = sin(timeSeconds * 5.0f);

  pushMatrix();
  pushStyle();
  translate(offsetX, offsetY);
  scale(scaleFactor);

  if (state == MEASURING) {
    drawMeasurementPhase();
  } else {
    drawTensileTestPhase();
  }

  popStyle();
  popMatrix();
}

void updateScreenScale() {
  scaleFactor = min((float)width / LOGICAL_WIDTH, (float)height / LOGICAL_HEIGHT);
  if (scaleFactor < 0.01f) scaleFactor = 1.0f;
  offsetX = (width - LOGICAL_WIDTH * scaleFactor) / 2.0f;
  offsetY = (height - LOGICAL_HEIGHT * scaleFactor) / 2.0f;
}

void drawMeasurementPhase() {
  // Smooth animations for Vernier caliper
  calX = lerp(calX, targetX, 0.1f);
  calY = lerp(calY, targetY, 0.1f);
  calRot = lerp(calRot, targetRot, 0.1f);
  calJaw = lerp(calJaw, targetJaw, 0.1f);

  // Draw Titles
  fill(227, 179, 65); textAlign(LEFT, TOP); textSize(24);
  text("PHASE 1: INITIAL MEASUREMENTS", 50, 30);
  fill(180); textSize(14);
  text("Use the Vernier Caliper to measure the specimen's diameter at 3 points, then the gauge length.", 50, 65);

  drawLargeSpecimen();
  drawNotebook();
  drawVernierCaliper(calX, calY, calRot, calJaw);
  
  btnMeasure.display();
}

void drawLargeSpecimen() {

  beginShape();
  // Right profile
  vertex(530, 20); vertex(530, 100);
  bezierVertex(530, 130, 500, 140, 500, 150); // curve to gauge
  vertex(500, 450); // right gauge
  bezierVertex(500, 460, 530, 470, 530, 500); // curve to bot grip
  vertex(530, 580); 
  // Left profile
  vertex(370, 580); vertex(370, 500);
  bezierVertex(370, 470, 400, 460, 400, 450); // curve to left gauge
  vertex(400, 150); // left gauge
  bezierVertex(400, 140, 370, 130, 370, 100); // curve to top grip
  vertex(370, 20);
  endShape(CLOSE);

  // Subtle Centerline
  stroke(180, 185, 190, 80); strokeWeight(1);
  line(450, 20, 450, 580);
}

void drawVernierCaliper(float x, float y, float rot, float jawDist) {
  pushMatrix();
  translate(x, y);
  rotate(rot);
  
  // Main scale bar
  fill(220); stroke(100); strokeWeight(2);
  rect(-180, -20, 500, 40, 4);
  
  // Fixed jaw
  rect(-40, 20, 40, 120, 2);
  
  // Main scale ticks
  fill(0); noStroke();
  for(int i=0; i<=300; i+=10) {
      rect(i, -20, 2, 15);
      if(i%50==0) {
          textSize(10); textAlign(CENTER, TOP);
          text(i/10, i, -5);
      }
  }
  
  // Sliding Part
  pushMatrix();
  translate(jawDist, 0);
  fill(235); stroke(100); strokeWeight(2);
  rect(-20, -25, 120, 50, 4);
  // Sliding jaw
  rect(0, 20, 40, 120, 2);
  
  // Vernier ticks
  fill(255, 0, 0); noStroke();
  for(int i=0; i<10; i++) {
      rect(i*9, 15, 1.5f, 10);
  }
  popMatrix();
  popMatrix();
}

void drawTensileTestPhase() {
  if (fractureFlash > 0) { fractureFlash -= 4.0f; }
  if (fractureFlash < 0) { fractureFlash = 0; }

  drawGraphPanel();
  drawTheoreticalCurve();
  drawActiveCurve();
  drawGraphMarkers();
  drawSpecimen();
  drawInstructor();
  drawStatusPanel();
  
  btnStart.display();
  btnInterrupt.display();
  btnReset.display();

  updatePhysics();
}

float calculateTheoreticalStress(float e) {
  if (e <= 0) return 0;
  if (e <= ELASTIC_LIMIT) return e * ELASTIC_SLOPE;
  if (e <= UTS_STRAIN) {
    float x = (e - ELASTIC_LIMIT) / (UTS_STRAIN - ELASTIC_LIMIT);
    return YIELD_STRESS + (PEAK_STRESS - YIELD_STRESS) * x * x;
  }
  if (e <= FRACTURE_STRAIN) {
    float x = (e - UTS_STRAIN) / (FRACTURE_STRAIN - UTS_STRAIN);
    return PEAK_STRESS - 55.0f * pow(x, 1.25f);
  }
  return 0;
}

float getFastStress(float e) {
  if (e <= 0) return 0;
  if (e >= FRACTURE_STRAIN) return cachedStress[(int)FRACTURE_STRAIN];
  int lower = floor(e);
  if (lower < 0) lower = 0;
  if (lower >= cachedStress.length - 1) lower = cachedStress.length - 2;
  float fraction = e - lower;
  return lerp(cachedStress[lower], cachedStress[lower + 1], fraction);
}

void updatePhysics() {
  if (state == LOADING) {
    strain += LOAD_SPEED;

    if (strain >= FRACTURE_STRAIN) {
      strain = FRACTURE_STRAIN;
      stress = getFastStress(strain);
      state = FRACTURED;
      tracePath.add(new TracePoint(strain, stress));
      fractureFlash = 255; specimenShake = 15;
      return;
    }

    stress = getFastStress(strain);
    maxStrainReached = max(maxStrainReached, strain);

    if (strain > UTS_STRAIN) {
      specimenShake = sin(timeSeconds * 20.0f) * ((strain - UTS_STRAIN) / 100.0f);
    }

  } else if (state == UNLOADING) {
    float deltaStrain = UNLOAD_SPEED;
    float elasticStressReduction = deltaStrain * ELASTIC_SLOPE;

    strain -= deltaStrain;
    stress -= elasticStressReduction;

    if (stress <= 0 || strain <= unloadTargetStrain) {
      stress = 0;
      strain = max(0, unloadTargetStrain);
      state = IDLE;
      tracePath.add(new TracePoint(strain, stress));
    }

  } else if (state == RELOADING) {
    strain += LOAD_SPEED;
    stress += LOAD_SPEED * ELASTIC_SLOPE;

    if (strain >= maxStrainReached) {
      strain = maxStrainReached;
      stress = getFastStress(strain);
      state = LOADING;
      tracePath.add(new TracePoint(strain, stress));
    }
  }

  if (specimenShake != 0) {
    specimenShake *= 0.90f;
    if (abs(specimenShake) < 0.05f) specimenShake = 0;
  }

  if (state != IDLE && state != FRACTURED) {
    TracePoint lastPt = tracePath.get(tracePath.size() - 1);
    if (lastPt == null || dist(lastPt.e, lastPt.s, strain, stress) > 0.5f) {
      tracePath.add(new TracePoint(strain, stress));
    }
  }
}

// --- Graph Rendering ---
void drawGraphPanel() {
  stroke(220); strokeWeight(2); noFill();

  // Axes
  line(graphX, graphY, graphX, graphY - graphHeight);
  line(graphX, graphY, graphX + graphWidth, graphY);

  // Grid
  stroke(255, 18); strokeWeight(1);
  for (int y = 0; y <= 350; y += 50) line(graphX, graphY - y, graphX + graphWidth, graphY - y);
  for (int x = 0; x <= 400; x += 50) line(graphX + x, graphY, graphX + x, graphY - graphHeight);

  // Labels
  fill(255); noStroke(); textAlign(RIGHT, CENTER); textSize(16);
  text("STRESS", graphX - 15, graphY - graphHeight + 10);
  textAlign(CENTER, TOP);
  text("STRAIN", graphX + graphWidth / 2, graphY + 15);

  fill(227, 179, 65); textAlign(LEFT, CENTER); textSize(18);
  text("PHASE 2: TENSILE TEST", graphX + 80, graphY - graphHeight + 25);

  // Elastic divider
  stroke(100, 200, 255, 160);
  dashedLine(graphX + ELASTIC_LIMIT, graphY, graphX + ELASTIC_LIMIT, graphY - graphHeight, 6);

  fill(100, 200, 255, 180); textAlign(CENTER, CENTER); textSize(13); noStroke();
  text("ELASTIC\nREGION", graphX + ELASTIC_LIMIT / 2, graphY - 300);
  fill(100, 255, 150, 180);
  text("PLASTIC\nREGION", graphX + 220, graphY - 300);
}

void drawTheoreticalCurve() {
  noFill(); stroke(255, 35); strokeWeight(2);
  beginShape();
  for (int e = 0; e <= FRACTURE_STRAIN; e += 2) vertex(graphX + e, graphY - cachedStress[e]);
  endShape();
}

void drawActiveCurve() {
  if (tracePath.size() == 0) return;
  noFill(); stroke(0, 255, 255); strokeWeight(3);
  beginShape();
  for (TracePoint pt : tracePath) vertex(graphX + pt.e, graphY - pt.s);
  vertex(graphX + strain, graphY - stress);
  endShape();

  noStroke(); fill(255);
  ellipse(graphX + strain, graphY - stress, 11, 11);

  if (state == LOADING && strain > 5 && strain <= ELASTIC_LIMIT) {
    fill(227, 179, 65, 35); stroke(227, 179, 65, 120); strokeWeight(1);
    rect(graphX + 115, graphY - 195, 170, 85, 8);
    fill(227, 179, 65); noStroke(); textAlign(CENTER, CENTER); textSize(14);
    text("ELASTIC REGION", graphX + 200, graphY - 175);
    fill(255); textSize(23); text("σ = Eε", graphX + 200, graphY - 143);
  }
}

void drawGraphMarkers() {
  drawMarker(ELASTIC_LIMIT, "Yield", 100, color(100, 200, 255));
  drawMarker(UTS_STRAIN, "UTS", PEAK_STRESS, color(255, 180, 80));
  drawMarker(FRACTURE_STRAIN, "Fracture", getFastStress(FRACTURE_STRAIN), color(255, 80, 80));
}

void drawMarker(float e, String label, float s, int c) {
  if (e > graphWidth) return;
  stroke(red(c), green(c), blue(c), 130); strokeWeight(1);
  dashedLine(graphX + e, graphY, graphX + e, graphY - min(s, graphHeight - 10), 5);
  noStroke(); fill(c); ellipse(graphX + e, graphY - s, 7, 7);
  textAlign(CENTER, TOP); textSize(10); text(label, graphX + e, graphY - 40);
}

// --- Specimen Visualization ---
void drawSpecimen() {
  float specX = 720, specY = 255;
  float baseWidth = 52, baseLength = 165;
  float currentLength = baseLength + strain * 0.43f;
  float topY = specY - currentLength / 2.0f;
  float botY = specY + currentLength / 2.0f;
  
  float necking = 0;
  if (maxStrainReached > UTS_STRAIN) {
    necking = (maxStrainReached - UTS_STRAIN) * 0.29f;
    necking = constrain(necking, 0, baseWidth * 0.42f);
  }

  float shakeX = (state == FRACTURED) ? sin(timeSeconds * 30) * specimenShake : 0;
  specX += shakeX;

  // Grips
  noStroke(); fill(40, 46, 54);
  rect(specX - 42, topY - 22, 84, 22, 4); rect(specX - 42, botY, 84, 22, 4);

  float centerY = (topY + botY) / 2.0f;
  if (state == FRACTURED) drawFracturedSpecimen(specX, topY, botY, centerY, baseWidth, necking);
  else drawNormalSpecimen(specX, topY, botY, centerY, baseWidth, necking);

  if (state == LOADING || state == RELOADING) drawForceArrows(specX, topY, botY);
}

void drawNormalSpecimen(float x, float topY, float botY, float centerY, float w, float neck) {
  stroke(150, 160, 170); strokeWeight(2); fill(225, 232, 238);
  beginShape();
  vertex(x + w / 2, topY);
  bezierVertex(x + w / 2, topY + 50, x + w / 2 - neck, centerY - 35, x + w / 2 - neck, centerY);
  bezierVertex(x + w / 2 - neck, centerY + 35, x + w / 2, botY - 50, x + w / 2, botY);
  vertex(x - w / 2, botY);
  bezierVertex(x - w / 2, botY - 50, x - w / 2 + neck, centerY + 35, x - w / 2 + neck, centerY);
  bezierVertex(x - w / 2 + neck, centerY - 35, x - w / 2, topY + 50, x - w / 2, topY);
  endShape(CLOSE);
}

void drawFracturedSpecimen(float x, float topY, float botY, float centerY, float w, float neck) {
  float gap = 18; stroke(150, 160, 170); strokeWeight(2); fill(225, 232, 238);
  // Top half
  beginShape();
  vertex(x - w / 2, topY); vertex(x + w / 2, topY);
  vertex(x + w / 2, centerY - gap - 30); vertex(x + w / 2 - neck, centerY - gap - 5);
  vertex(x + w / 2 - neck - 5, centerY - gap + 3); vertex(x - w / 2 + neck + 5, centerY - gap - 3);
  vertex(x - w / 2, centerY - gap - 30);
  endShape(CLOSE);
  // Bottom half
  beginShape();
  vertex(x - w / 2, botY); vertex(x + w / 2, botY);
  vertex(x + w / 2, centerY + gap + 30); vertex(x + w / 2 - neck - 5, centerY + gap + 3);
  vertex(x - w / 2 + neck + 5, centerY + gap - 3); vertex(x - w / 2, centerY + gap + 30);
  endShape(CLOSE);
  // Crack
  stroke(255, 90, 90); line(x - 18, centerY - gap, x - 5, centerY - gap + 5);
  line(x - 5, centerY - gap + 5, x + 7, centerY - gap - 4);
  line(x + 7, centerY - gap - 4, x + 20, centerY - gap + 4);
}

void drawForceArrows(float x, float topY, float botY) {
  float pulse = arrowPulse * 3; stroke(60, 200, 90); strokeWeight(3); fill(60, 200, 90);
  line(x, topY - 30 - pulse, x, topY - 60 - pulse);
  triangle(x, topY - 72 - pulse, x - 10, topY - 55 - pulse, x + 10, topY - 55 - pulse);
  line(x, botY + 30 + pulse, x, botY + 60 + pulse);
  triangle(x, botY + 72 + pulse, x - 10, botY + 55 + pulse, x + 10, botY + 55 + pulse);
}

// --- Status, Dashboard, and Instructor ---
void drawInstructor() {
  float x = 845, y = 395;
  String line1 = "Press Apply Load", line2 = "";

  if (state == LOADING) {
    if (strain < ELASTIC_LIMIT) { line1 = "Elastic Region"; line2 = "σ = Eε"; }
    else if (strain < UTS_STRAIN) { line1 = "Plastic Region"; line2 = "Permanent deform"; }
    else { line1 = "Necking"; line2 = "Area is reducing"; }
  } else if (state == UNLOADING) { line1 = "Unloading"; line2 = (unloadTargetStrain < 2) ? "Returns to origin" : "Permanent set!"; }
  else if (state == RELOADING) { line1 = "Reloading"; line2 = "Elastic recovery"; }
  else if (state == FRACTURED) { line1 = "FRACTURE!"; line2 = "Specimen failed"; }
  else if (strain > 0) { line1 = "Load again"; line2 = "Observe the curve"; }

  // Bubble ONLY (Stickman removed as requested)
  noStroke(); fill(255); rect(x - 155, y - 115, 155, 58, 10);
  triangle(x - 70, y - 57, x - 50, y - 57, x - 40, y - 35);
  fill(20); textAlign(CENTER, CENTER); textSize(12);
  text(line1, x - 77, y - 98); text(line2, x - 77, y - 78);
}

void drawStatusPanel() {
  float x = 555, y = 30;
  fill(23, 28, 35); noStroke(); rect(x, y, 320, 160, 10);

  // State string
  String stateStr = "IDLE";
  int stColor = color(255);
  if (state == LOADING) { stateStr = "LOADING"; stColor = color(70,220,110); }
  else if (state == UNLOADING) { stateStr = "UNLOADING"; stColor = color(255,100,100); }
  else if (state == RELOADING) { stateStr = "RELOADING"; stColor = color(220,220,70); }
  else if (state == FRACTURED) { stateStr = "FRACTURED"; stColor = color(255,60,60); }
  
  fill(180); textAlign(LEFT, CENTER); textSize(12); text("STATE", x + 15, y + 20);
  fill(stColor); textSize(17); text(stateStr, x + 90, y + 20);

  // Physical mapping
  float physStrain = strain / 1000.0f;
  float initL = 50.00f;
  float initD = 12.50f;
  float deltaL = physStrain * initL;
  
  // Poisson's ratio logic
  float nu = 0.3f;
  if (strain > ELASTIC_LIMIT) {
    float plasticRatio = constrain((strain - ELASTIC_LIMIT) / 100.0f, 0, 1);
    nu = lerp(0.3f, 0.45f, plasticRatio);
  }
  float deltaD = -nu * physStrain * initD;

  // Row 1: Stress / Strain
  fill(180); textSize(12);
  text("Stress:", x + 15, y + 55); fill(255); textSize(14); text(nf(stress, 0, 1) + " MPa", x + 70, y + 55);
  fill(180); textSize(12);
  text("Strain:", x + 175, y + 55); fill(255); textSize(14); text(nf(physStrain, 0, 3), x + 225, y + 55);

  // Row 2: L0 / D0
  fill(150, 200, 255); textSize(13);
  text("L0:  50.00 mm", x + 15, y + 85);
  text("D0:  12.50 mm", x + 175, y + 85);

  // Row 3: del L / del D
  fill(255, 170, 170);
  String signL = deltaL >= 0 ? "+" : "-";
  String signD = deltaD >= 0 ? "+" : "-";
  text("ΔL: " + signL + nf(abs(deltaL), 0, 2) + " mm", x + 15, y + 110);
  text("Δd: " + signD + nf(abs(deltaD), 0, 3) + " mm", x + 175, y + 110);

  // Row 4: Live L / Live D
  fill(150, 255, 150);
  text("L:  " + nf(initL + deltaL, 0, 2) + " mm", x + 15, y + 135);
  text("d:  " + nf(initD + deltaD, 0, 2) + " mm", x + 175, y + 135);
}

// ================================================================
// INPUT HANDLING
// ================================================================
void mousePressed() {
  if (scaleFactor <= 0.001f) return;
  float logicalX = (mouseX - offsetX) / scaleFactor;
  float logicalY = (mouseY - offsetY) / scaleFactor;

  if (state == MEASURING) {
    if (btnMeasure.isHit(logicalX, logicalY)) {
      measStep++;
      if (measStep == 1) { 
        targetX = 400; targetY = 300; targetRot = 0; targetJaw = 100;
        btnMeasure.label = "MEASURE d2 (Mid)";
      } else if (measStep == 2) { 
        targetY = 420; 
        btnMeasure.label = "MEASURE d3 (Bot)";
      } else if (measStep == 3) { 
        targetX = 510; targetY = 150; targetRot = PI/2; targetJaw = 300;
        btnMeasure.label = "MEASURE Gauge L0";
      } else if (measStep == 4) { 
        targetX = 650; targetY = 150; targetRot = PI/2; targetJaw = 0;
        btnMeasure.label = "START TENSILE TEST";
      } else if (measStep == 5) {
        state = IDLE; 
      }
    }
    return;
  }

  // APPLY LOAD CLICKED (Phase 2)
  if (btnStart.isHit(logicalX, logicalY)) {
    if (state == IDLE || state == UNLOADING) {
      if (strain < maxStrainReached - 0.1f) {
        state = RELOADING;
        unloadTargetStrain = strain - stress / ELASTIC_SLOPE;
      } else {
        state = LOADING;
      }
    }
    return;
  }

  // UNLOAD (INTERRUPT) CLICKED (Phase 2)
  if (btnInterrupt.isHit(logicalX, logicalY)) {
    if ((state == LOADING || state == RELOADING) && strain > 0) {
      state = UNLOADING;
      unloadTargetStrain = strain - stress / ELASTIC_SLOPE;
    }
    return;
  }

  // RESET CLICKED
  if (btnReset.isHit(logicalX, logicalY)) resetSimulation();
}

void resetSimulation() {
  state = MEASURING; measStep = 0;
  strain = 0; stress = 0; maxStrainReached = 0; unloadTargetStrain = 0;
  tracePath.clear(); tracePath.add(new TracePoint(0, 0));
  fractureFlash = 0; specimenShake = 0;
  targetX = 650; targetY = 80; targetRot = 0; targetJaw = 250;
  calX = 650; calY = 80; calRot = 0; calJaw = 250;
  btnMeasure.label = "MEASURE d1 (Top)";
}

class Button {
  String label;
  float x, y, w, h;
  int buttonColor;
  
  Button(String label, float x, float y, float w, float h, int c) {
    this.label = label; this.x = x; this.y = y; this.w = w; this.h = h; this.buttonColor = c;
  }
  
  void display() {
    noStroke(); fill(0, 100); rect(this.x + 2, this.y + 3, this.w, this.h, 7);
    fill(this.buttonColor); rect(this.x, this.y, this.w, this.h, 7);
    fill(255, 30); rect(this.x, this.y, this.w, 15, 7);
    fill(255); textAlign(CENTER, CENTER); textSize(14);
    text(this.label, this.x + this.w / 2, this.y + this.h / 2);
  }
  
  boolean isHit(float mx, float my) { 
    return mx >= this.x && mx <= this.x + this.w && my >= this.y && my <= this.y + this.h; 
  }
}

void dashedLine(float x1, float y1, float x2, float y2, float dashLength) {
  float distance = dist(x1, y1, x2, y2);
  if (distance <= 0 || dashLength <= 0) return;
  int segments = max(1, ceil(distance / dashLength));
  for (int i = 0; i < segments; i += 2) {
    float sx = lerp(x1, x2, (float)i / segments);
    float sy = lerp(y1, y2, (float)i / segments);
    float ex = lerp(x1, x2, min((float)(i + 1) / segments, 1.0f));
    float ey = lerp(y1, y2, min((float)(i + 1) / segments, 1.0f));
    line(sx, sy, ex, ey);
  }
}
