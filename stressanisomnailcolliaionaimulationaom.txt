void calculatePhysics() {
  A = PI * D * D / 4.0;
  staticStress = P / A;
  staticExtension = P * L / (A * E);
  impactVelocity = sqrt(2.0 * G * H);
  
  float insideRoot = 1.0 + 2.0 * H / staticExtension;
  impactFactor = 1.0 + sqrt(insideRoot);
  
  sigmaMax = staticStress * impactFactor;
  FMAX = sigmaMax * A;
  maxExtension = sigmaMax * L / E;
  maxStrain = sigmaMax / E;
  
  potentialEnergy = P * H;
  kineticEnergy = potentialEnergy;
  strainEnergy = sigmaMax * sigmaMax * A * L / (2.0 * E);
  
  float EPa = E * 1000000.0;
  waveSpeed = sqrt(EPa / rho);
  waveSpeedMmPerMs = waveSpeed;
  waveTravelTime = L / waveSpeedMmPerMs;
  
  velocityBeforeImpact = impactVelocity;
  velocityAfterImpact = -restitutionCoefficient * velocityBeforeImpact;
}

void updatePhysics() {
  if (state == FALLING) {
    float animationGravity = visualGravity * fallAnimationSpeed;
    weightVelocity += animationGravity;
    weightY += weightVelocity;
    
    if (weightY >= impactTargetWeightY) {
      weightY = impactTargetWeightY;
      velocityBeforeImpact = impactVelocity;
      velocityAfterImpact = -restitutionCoefficient * velocityBeforeImpact;
      weightVelocity = 0.0;
      
      state = IMPACT;
      impactTimeMs = 0.0;
      waveDistance = 0.0;
      redPulse = 0.0;
      contactSweepX = -nailHeadRadius;
      contactActive = true;
    }
  }
  else if (state == IMPACT || state == WAVE) {
    float frameMs = 1000.0 / max(1.0, frameRate);
    float impactDt = frameMs * impactAnimationSpeed;
    
    float currentEnvelope = forceEnvelope(impactTimeMs);
    redPulse = currentEnvelope;
    weightVelocity = 0.0;
    
    float penetrationStep = currentEnvelope * impactDt * 0.015;
    totalPenetration += penetrationStep;
    weightY = impactTargetWeightY + totalPenetration;
    
    impactTimeMs += impactDt;
    
    if (impactTimeMs <= pulseEndMs) {
      float progress = constrain(impactTimeMs / pulseEndMs, 0.0, 1.0);
      contactSweepX = map(progress, 0.0, 1.0, -nailHeadRadius, nailHeadRadius);
      contactActive = true;
    } else {
      contactActive = false;
      contactSweepX = nailHeadRadius;
    }
    
    waveDistance = min(L, impactTimeMs * waveSpeedMmPerMs);
    
    if (state == IMPACT && impactTimeMs > pulseEndMs) {
      state = WAVE;
    }
    
    if (state == WAVE && waveDistance >= L) {
      state = SETTLED;
    }
  }
  else if (state == SETTLED) {
    weightVelocity = 0.0;
    contactActive = false;
    waveDistance = L;
    redPulse *= 0.96;
  }
}

float forceEnvelope(float t) {
  if (t <= 0.0 || t > pulseEndMs) return 0.0;
  float ratio = t / peakTimeMs;
  float value = ratio * exp(1.0 - ratio);
  return constrain(value, 0.0, 1.0);
}

// ================================================================
// FEA JET/RAINBOW COLORMAP ENGINE
// ================================================================

color getSmoothHeatMap(float rawIntensity) {
  float intensity = constrain(rawIntensity, 0.0, 1.0);
  
  // Exact FEA (Von Mises) Rainbow Color Interpolation
  if (intensity < 0.05) {
    // 0.00 - 0.05: Transition out of Steel base into Deep Blue
    return lerpColor(steelColor, color(0, 0, 150), intensity / 0.05);
  } else if (intensity < 0.25) {
    // 0.05 - 0.25: Deep Blue to Cyan
    return lerpColor(color(0, 0, 150), color(0, 200, 255), (intensity - 0.05) / 0.20);
  } else if (intensity < 0.50) {
    // 0.25 - 0.50: Cyan to Bright Green
    return lerpColor(color(0, 200, 255), color(0, 220, 0), (intensity - 0.25) / 0.25);
  } else if (intensity < 0.75) {
    // 0.50 - 0.75: Bright Green to Yellow
    return lerpColor(color(0, 220, 0), color(255, 220, 0), (intensity - 0.50) / 0.25);
  } else if (intensity < 0.90) {
    // 0.75 - 0.90: Yellow to Orange
    return lerpColor(color(255, 220, 0), color(255, 100, 0), (intensity - 0.75) / 0.15);
  } else {
    // 0.90 - 1.00: Orange to Deep Red (Max Stress)
    return lerpColor(color(255, 100, 0), color(200, 0, 0), (intensity - 0.90) / 0.10);
  }
}

void drawStressNail(float radius, float h, int sides) {
  noStroke();
  beginShape(QUAD_STRIP);
  for (int i = 0; i <= sides; i++) {
    float angle = TWO_PI * i / sides;
    float x = cos(angle) * radius;
    float z = sin(angle) * radius;
    fill(stressColorAtDistance(0));
    vertex(x, 0, z);
    fill(stressColorAtDistance(h));
    vertex(x, h, z);
  }
  endShape();
  drawNailGrid(radius, h, sides);
}

void drawNailGrid(float radius, float h, int sides) {
  strokeWeight(1.3);
  for (int i = 0; i < 16; i++) {
    float angle = TWO_PI * i / 16.0;
    float x = cos(angle) * radius * 1.01;
    float z = sin(angle) * radius * 1.01;
    stroke(80, 100, 125, 180);
    line(x, 0, z, x, h, z);
  }
  for (int j = 0; j <= 30; j++) {
    float y = h * j / 30.0;
    drawRingXZ(0, y, radius * 2.02, stressColorAtDistance(y));
  }
}

color stressColorAtDistance(float distance) {
  if (state < IMPACT) return steelColor;
  
  float propagationTime = distance / waveSpeedMmPerMs;
  float localTime = impactTimeMs - propagationTime;
  float envelope = forceEnvelope(localTime);
  float decay = exp(-distance / decayLength);
  float intensity = envelope * decay;
  
  if (localTime > pulseEndMs) {
    float residual = exp(-(localTime - pulseEndMs) / 10.0) * decay;
    intensity = max(intensity, residual * 0.55);
  }
  
  // Apply Stress Multiplier scale
  return getSmoothHeatMap(intensity * stressIntensity);
}

void drawContactForceCurve() {
  if (!contactActive) return;
  pushMatrix();
  translate(0, contactWorldY - (nailWorldBaseY + totalPenetration), 0);
  
  float progress = constrain(impactTimeMs / pulseEndMs, 0.0, 1.0);
  float centerX = map(progress, 0.0, 1.0, -nailHeadRadius, nailHeadRadius);
  
  noFill(); stroke(getSmoothHeatMap(redPulse * stressIntensity), 255); strokeWeight(3 + 5 * redPulse);
  beginShape();
  int samples = 120;
  float visibleWidth = nailHeadRadius * contactCurveWidth / 18.0;
  for (int i = 0; i <= samples; i++) {
    float u = map(i, 0, samples, -1.0, 1.0);
    float x = centerX + u * visibleWidth;
    float localTime = impactTimeMs + u * peakTimeMs;
    float envelope = forceEnvelope(localTime);
    float curveY = -envelope * contactCurveHeight;
    x = constrain(x, -nailHeadRadius, nailHeadRadius);
    vertex(x, curveY, 0);
  }
  endShape();
  popMatrix();
}

void drawContactReferencePoint() {

  color dynamicColor = getSmoothHeatMap(redPulse * stressIntensity);
  noStroke(); fill(dynamicColor, 180 + 75 * blink);
  sphereDetail(12); sphere(5 + 7 * redPulse);

}

void drawContactGlow() {
  if (!contactActive) return;
  pushMatrix();
  translate(contactSweepX, contactWorldY - (nailWorldBaseY + totalPenetration) - 2, 0);
  float blinkFrequency = 0.01 + 0.01 * stressIntensity; // BLINK SPEED MULTIPLIER
  float blink = 0.5 + 0.5 * sin(millis() * blinkFrequency);
  color glowColor = getSmoothHeatMap(redPulse * stressIntensity);
  for (int i = 0; i < 5; i++) {
    float diameter = 20 + i * 14;
    stroke(glowColor, (130 - i * 22) * redPulse * blink);
    strokeWeight(max(1, 4 - i * 0.5));
    noFill(); ellipseXZ(0, 0, diameter);
  }
  popMatrix();
}

void drawStressWave() {
  if (state < IMPACT) return;
  pushMatrix();
  translate(0, nailWorldBaseY + totalPenetration, 0);
  
  float waveY = constrain(waveDistance, 0, visualNailLength);
  float localTime = impactTimeMs - waveY / waveSpeedMmPerMs;
  float localEnvelope = forceEnvelope(localTime);
  float localDecay = exp(-waveY / decayLength);
  float intensity = constrain(localEnvelope * localDecay * waveVisualScale, 0.0, 1.0);
  
  color waveColor = getSmoothHeatMap(intensity * stressIntensity);
  
  noFill();
  stroke(waveColor, 230 * max(0.15, intensity)); strokeWeight(5); drawRingXZ(0, waveY, 60);
  stroke(hotWhite, 160 * max(0.15, intensity)); strokeWeight(2); drawRingXZ(0, waveY, 90);
  popMatrix();
}

void drawImpulseRings() {
  pushMatrix();
  translate(0, nailWorldBaseY + totalPenetration, 0);
  for (int i = 0; i < 12; i++) {
    float distance = waveDistance - i * 28;
    if (distance < 0) continue;
    float alpha = 220 - i * 16;
    float radius = 25 + i * 8;
    float timeAtDistance = impactTimeMs - distance / waveSpeedMmPerMs;
    float intensity = forceEnvelope(timeAtDistance) * exp(-distance / decayLength);
    drawRingXZ(0, distance, radius, color(getSmoothHeatMap(intensity * stressIntensity), max(0, alpha)));
  }
  popMatrix();
}

void drawStressVectors() {
  pushMatrix();
  translate(0, nailWorldBaseY + totalPenetration, 0);
  for (int i = 0; i < 12; i++) {
    float distance = i * (L / 12.0);
    float timeAtDistance = impactTimeMs - distance / waveSpeedMmPerMs;
    float envelope = forceEnvelope(timeAtDistance);
    float decay = exp(-distance / decayLength);
    float strength = envelope * decay;
    if (strength > 0.01) {
      float visualY = map(distance, 0, L, 0, visualNailLength);
      drawDownVector(0, visualY, 0, 80 * strength, strength);
    }
  }
  popMatrix();
}

  pushMatrix(); translate(x, y + length, z);
  noStroke(); fill(vectorColor, 255 * intensity); drawDownArrowHead(22, 12, 5);
  popMatrix();
}

  vertex(0, tipLength, 0); vertex(baseWidth, 0, -depth); vertex(baseWidth, 0, depth);
  endShape();
}

void drawForceTimeGraph() {
  fill(panelColor); noStroke(); rectMode(CORNER);
  rect(graphX, graphY, graphW, graphH, 16);
  fill(uiRed); textAlign(LEFT, TOP); textFont(boldFont); textSize(19);
  text("IMPULSIVE FORCE — F(t)", graphX + 18, graphY + 16);
  fill(150, 170, 190); textFont(regularFont); textSize(12); textAlign(RIGHT, TOP);
  text("DRAG PANEL", graphX + graphW - 15, graphY + 18);
  
  int px = int(graphX) + 45, py = int(graphY) + 55, pw = int(graphW) - 75, ph = int(graphH) - 90;
  stroke(120, 135, 155); strokeWeight(1.5);
  line(px, py + ph, px + pw, py + ph); line(px, py, px, py + ph);
  
  stroke(50, 65, 80);
  for (int i = 1; i < 5; i++) { float xx = px + pw * i / 5.0; line(xx, py, xx, py + ph); }
  for (int i = 1; i < 4; i++) { float yy = py + ph * i / 4.0; line(px, yy, px + pw, yy); }
  
  noFill(); stroke(uiRed); strokeWeight(4); beginShape();
  int samples = 160;
  for (int i = 0; i <= samples; i++) {
    float t = pulseEndMs * i / float(samples);
    float value = forceEnvelope(t);
    float xx = px + map(t, 0, pulseEndMs, 0, pw);
    float yy = py + ph - value * ph;
    vertex(xx, yy);
  }
  endShape();
  
  stroke(uiYellow, 220); strokeWeight(2); line(px, py + ph - 4, px + pw, py + ph - 4);
  float peakX = px + map(peakTimeMs, 0, pulseEndMs, 0, pw);
  stroke(uiYellow, 160); strokeWeight(1.5); line(peakX, py, peakX, py + ph);
  
  if (contactActive) {
    float currentForce = forceEnvelope(impactTimeMs);
    float cursorX = px + map(impactTimeMs, 0, pulseEndMs, 0, pw);
    float currentY = py + ph - currentForce * ph;
    
    // Graph cursor color also scales with the stress multiplier
    color cursorColor = getSmoothHeatMap(currentForce * stressIntensity);
    stroke(cursorColor); strokeWeight(2); line(cursorX, py, cursorX, py + ph);
    noStroke(); fill(cursorColor); ellipse(cursorX, currentY, 13, 13);
    
    float blinkFrequency = 0.01 + 0.01 * stressIntensity;
    float blink = 0.5 + 0.5 * sin(millis() * blinkFrequency);
    fill(hotWhite, 170 + 85 * blink); ellipse(cursorX, currentY, 9 + 7 * blink, 9 + 7 * blink);
  }
  
  fill(200); textFont(regularFont); textSize(14); textAlign(CENTER, TOP);
  text("0 ms", px, py + ph + 10); text("3 ms", px + pw / 2, py + ph + 10); text("6 ms", px + pw, py + ph + 10);
  fill(uiYellow); textAlign(LEFT, TOP);
  text("PEAK = " + nf(peakTimeMs, 0, 1) + " ms", peakX + 5, py + 5);
  fill(uiRed);
  text("FMAX = " + nf(FMAX / 1000.0, 0, 2) + " kN", graphX + 18, graphY + graphH - 28);
}

void startDrop() {
  calculateContactGeometry(); calculatePhysics();
  state = FALLING; weightY = 0.0; weightVelocity = 0.0; impactTimeMs = 0.0;
  waveDistance = 0.0; redPulse = 0.0; totalPenetration = 0.0; contactSweepX = -nailHeadRadius;
  contactActive = false; velocityBeforeImpact = impactVelocity; velocityAfterImpact = 0.0;
}

boolean isInsideImpactPanel(float px, float py) { return (px >= impactPanelX && px <= impactPanelX + impactPanelW && py >= impactPanelY && py <= impactPanelY + impactPanelH); }
boolean isInsideStressPanel(float px, float py) { return (px >= stressPanelX && px <= stressPanelX + stressPanelW && py >= stressPanelY && py <= stressPanelY + stressPanelH); }

boolean isInsideVector(float px, float py) {
  if (!contactActive) return false;
  float forceFraction = forceEnvelope(impactTimeMs);
  if (forceFraction <= 0.001) return false;
  
  float vectorLength = impactVectorMinimum + (impactVectorMaximum - impactVectorMinimum) * forceFraction;
  float left = vectorScreenX - 45; float right = vectorScreenX + 45;
  float top = vectorScreenY - 25; float bottom = vectorScreenY + vectorLength + 45;
  
  return (px >= left && px <= right && py >= top && py <= bottom);
}

void mousePressed() {
  if (viewDataScreen) {
    if (mouseX >= backBtnX && mouseX <= backBtnX + backBtnW && mouseY >= backBtnY && mouseY <= backBtnY + backBtnH) {
      viewDataScreen = false;
    }
    return; 
  }
  
  if (mouseX >= viewDataBtnX && mouseX <= viewDataBtnX + viewDataBtnW && mouseY >= viewDataBtnY && mouseY <= viewDataBtnY + viewDataBtnH) {
    viewDataScreen = true;
    return;
  }
  
  if (isInsideDropButton(mouseX, mouseY)) { if (state == READY || state == SETTLED) startDrop(); return; }
  if (isInsideResetButton(mouseX, mouseY)) { resetSimulation(); return; }
  
  if (isInsideStressSlider(mouseX, mouseY)) { stressSliderDragging = true; updateStressFromMouse(); return; }
  if (isInsideFallSlider(mouseX, mouseY)) { fallSliderDragging = true; updateFallSpeedFromMouse(); return; }
  if (isInsideImpactSlider(mouseX, mouseY)) { impactSliderDragging = true; updateImpactSpeedFromMouse(); return; }
  
  if (isInsideVector(mouseX, mouseY)) {
    vectorDragging = true; vectorDetached = true;
    vectorDragOffsetX = mouseX - vectorScreenX; vectorDragOffsetY = mouseY - vectorScreenY;
    return;
  }
  
  if (isInsideStressPanel(mouseX, mouseY)) { stressPanelDragging = true; stressDragOffsetX = mouseX - stressPanelX; stressDragOffsetY = mouseY - stressPanelY; return; }
  if (isInsideFallPanel(mouseX, mouseY)) { fallPanelDragging = true; fallDragOffsetX = mouseX - fallPanelX; fallDragOffsetY = mouseY - fallPanelY; return; }
  if (isInsideImpactPanel(mouseX, mouseY)) { impactPanelDragging = true; impactDragOffsetX = mouseX - impactPanelX; impactDragOffsetY = mouseY - impactPanelY; return; }
  if (isInsideGraph(mouseX, mouseY)) { graphDragging = true; graphDragOffsetX = mouseX - graphX; graphDragOffsetY = mouseY - graphY; return; }
  
  cameraDragging = true; lastMouseX = mouseX; lastMouseY = mouseY;
}

void updateStressFromMouse() {
  float sliderX = stressPanelX + 40, sliderW = stressPanelW - 80;
  stressIntensity = constrain(map(mouseX, sliderX, sliderX + sliderW, minStress, maxStress), minStress, maxStress);
}

void updateFallSpeedFromMouse() {
  float sliderX = fallPanelX + 40, sliderW = fallPanelW - 80;
  fallAnimationSpeed = constrain(map(mouseX, sliderX, sliderX + sliderW, minFallSpeed, maxFallSpeed), minFallSpeed, maxFallSpeed);
}

void updateImpactSpeedFromMouse() {
  float sliderX = impactPanelX + 40, sliderW = impactPanelW - 80;
  impactAnimationSpeed = constrain(map(mouseX, sliderX, sliderX + sliderW, minImpactSpeed, maxImpactSpeed), minImpactSpeed, maxImpactSpeed);
}

void mouseDragged() {
  if (viewDataScreen) return; 
  
  if (stressSliderDragging) { updateStressFromMouse(); return; }
  if (fallSliderDragging) { updateFallSpeedFromMouse(); return; }
  if (impactSliderDragging) { updateImpactSpeedFromMouse(); return; }
  
  if (vectorDragging) { vectorScreenX = mouseX - vectorDragOffsetX; vectorScreenY = mouseY - vectorDragOffsetY; return; }
  
  if (stressPanelDragging) { stressPanelX = mouseX - stressDragOffsetX; stressPanelY = mouseY - stressDragOffsetY; constrainPanels(); return; }
  if (fallPanelDragging) { fallPanelX = mouseX - fallDragOffsetX; fallPanelY = mouseY - fallDragOffsetY; constrainPanels(); return; }
  if (impactPanelDragging) { impactPanelX = mouseX - impactDragOffsetX; impactPanelY = mouseY - impactDragOffsetY; constrainPanels(); return; }
  if (graphDragging) { graphX = mouseX - graphDragOffsetX; graphY = mouseY - graphDragOffsetY; constrainPanels(); return; }
  
  if (cameraDragging) {
    float dx = mouseX - lastMouseX; float dy = mouseY - lastMouseY;
    rotY += dx * 0.01; rotX += dy * 0.01; rotX = constrain(rotX, -PI / 2.2, PI / 2.2);
    lastMouseX = mouseX; lastMouseY = mouseY;
  }
}

void mouseReleased() {
  stressSliderDragging = false; stressPanelDragging = false;
  fallSliderDragging = false; fallPanelDragging = false;
  impactSliderDragging = false; impactPanelDragging = false;
  graphDragging = false; vectorDragging = false; cameraDragging = false;
}

// ================================================================
// TOUCH & KEYBOARD
// ================================================================

void touchStarted() { mousePressed(); }
void touchMoved() { mouseDragged(); }
void touchEnded() { mouseReleased(); }
void mouseWheel(float amount) { if(!viewDataScreen){ zoom *= 1.0 - amount * 0.05; zoom = constrain(zoom, 0.55, 2.5); } }

void keyPressed() {
  if (key == 'r' || key == 'R') resetSimulation();
  if (key == 'd' || key == 'D') if (state == READY || state == SETTLED) startDrop();
  if (key == '1') impactAnimationSpeed = 0.001;
  if (key == '2') impactAnimationSpeed = 0.050;
  if (key == '3') impactAnimationSpeed = 0.100;
  if (key == '4') impactAnimationSpeed = 0.250;
  if (key == '5') impactAnimationSpeed = 1.000;
}
