void drawSimulationScreen() {
  background(238, 242, 246); setupCamera(cameraDistance);
  lights(); ambientLight(110, 110, 110); directionalLight(255, 255, 255, -0.5, -1.0, -0.7);

  updateFalling(); updateImpact();
  drawGround(650, 500); drawHeight();

  currentSolid.drawCrossSection(sliceSlider.value);
  if (!falling) currentSolid.drawStress(impactProgress, currentStressMPa);
  
  currentSolid.drawTransparentShell();
  currentSolid.drawAnnotations();

  if (falling) drawFallingLoad(currentSolid.centerY - currentSolid.halfHeight());
  drawImpulseVector(true);
  drawSimulationHUD("3D IMPACT DROP");
}

// DROP ALL SCREEN
// ============================================================
void drawDropAllScreen() {
  background(238, 242, 246); 
  setupCamera(cameraDistance * 2.1); // Zoomed out for 3x3 grid
  lights(); ambientLight(110, 110, 110); directionalLight(255, 255, 255, -0.5, -1.0, -0.7);

  updateFallingAll(); updateImpact();
  drawGround(1600, 1000);

  // 3x3 Grid pattern for 9 shapes
  float[] xOff = {-450, 0, 450, -450, 0, 450, -450, 0, 450};
  float[] zOff = {-350, -350, -350, 0, 0, 0, 350, 350, 350};

  for (int i = 0; i < SHAPE_COUNT; i++) {
    Solid s = allSolids[i];
    pushMatrix(); translate(xOff[i], 0, zOff[i]);
    s.drawCrossSection(sliceSlider.value);
    if (!falling) s.drawStress(impactProgress, currentStressMPa);
    s.drawTransparentShell();
    if (falling) drawFallingLoad(s.centerY - s.halfHeight());
    drawImpulseVector(false);
    drawWorldText(s.name, 0, groundY + 30, 0, 22, color(40));
    popMatrix();
  }
  drawSimulationHUD("ALL SOLIDS IMPACT");
}

// RAMP JUMP SCREEN
// ============================================================
void drawJumpScreen() {
  background(238, 242, 246); setupCamera(cameraDistance * 1.6);
  lights(); ambientLight(110, 110, 110); directionalLight(255, 255, 255, -0.5, -1.0, -0.7);

  updateJump(); updateImpact();

  // Draw Ground and Ramp
  pushMatrix(); translate(0, groundY + 5.0, 0); fill(175, 180, 185); stroke(90); strokeWeight(2); box(2000, 10, 600); popMatrix();
  drawWorldText("LANDING ZONE", 400, groundY + 35, 230, 22, color(20));
  
  // Ramp
  pushMatrix(); translate((RAMP_START_X + RAMP_END_X)/2.0, groundY - RAMP_HEIGHT/2.0, 0);
  fill(150, 155, 160); noStroke();
  beginShape(TRIANGLES);
  vertex(-(RAMP_END_X - RAMP_START_X)/2.0, RAMP_HEIGHT/2.0, -300); vertex((RAMP_END_X - RAMP_START_X)/2.0, RAMP_HEIGHT/2.0, -300); vertex((RAMP_END_X - RAMP_START_X)/2.0, -RAMP_HEIGHT/2.0, -300);
  vertex(-(RAMP_END_X - RAMP_START_X)/2.0, RAMP_HEIGHT/2.0, 300); vertex((RAMP_END_X - RAMP_START_X)/2.0, RAMP_HEIGHT/2.0, 300); vertex((RAMP_END_X - RAMP_START_X)/2.0, -RAMP_HEIGHT/2.0, 300);
  endShape();
  beginShape(QUADS);
  vertex(-(RAMP_END_X - RAMP_START_X)/2.0, RAMP_HEIGHT/2.0, -300); vertex((RAMP_END_X - RAMP_START_X)/2.0, -RAMP_HEIGHT/2.0, -300); vertex((RAMP_END_X - RAMP_START_X)/2.0, -RAMP_HEIGHT/2.0, 300); vertex(-(RAMP_END_X - RAMP_START_X)/2.0, RAMP_HEIGHT/2.0, 300);
  endShape();
  popMatrix();

  // Draw Trail
  noFill(); stroke(0, 150, 255, 150); strokeWeight(4);
  beginShape(); for (PVector p : jumpTrail) vertex(p.x, p.y, p.z); endShape();

  // Draw Vehicle
  pushMatrix();
  translate(jumpX, jumpY, 0);
  rotateY(HALF_PI); // Rotate to face X axis (direction of travel)
  rotateX(jumpPitch); // Pitch up/down

  currentSolid.centerY = 0; // Local centering
  currentSolid.drawCrossSection(sliceSlider.value);
  if (jumpState == 3) currentSolid.drawStress(impactProgress, currentStressMPa);
  currentSolid.drawTransparentShell();
  
  if (jumpState == 3) {
    rotateX(-jumpPitch); rotateY(-HALF_PI); // unrotate for impulse vector
    pushMatrix(); translate(0, currentSolid.halfHeight(), 0); drawImpulseVector(true); popMatrix();
  }
  popMatrix();

  // Draw Metrics HUD
  hint(DISABLE_DEPTH_TEST); camera();
  fill(20); textSize(22); text(currentSolid.name + " - RAMP JUMP", width / 2, 30);
  resetButton.draw(); menuButton.draw();

  fill(255, 255, 255, 238); stroke(100); strokeWeight(2); rect(10, height - 140, width - 20, 120, 10);
  fill(0, 100, 200); textSize(16); text("JUMP ANALYTICS", width / 2, height - 125);
  fill(25); textSize(14);
  text("Max Height (h): " + nf(jumpMaxHeight, 1, 1) + " units", width / 2, height - 100);
  text("Jump Length (L): " + (jumpState==3 ? nf(jumpLength, 1, 1) : "---") + " units", width / 2, height - 80);
  text(currentSolid.areaText(), width / 2, height - 60);
  text("Landing Stress: " + (jumpState==3 ? nf(currentStressMPa, 1, 2) + " MPa" : "---"), width / 2, height - 40);

  if (jumpState == 3) {
    sliceSlider.draw("CROSS-SECTION SLICER"); 
    impactSlider.draw("IMPACT TIME / STRESS PROPAGATION");
  }
  hint(ENABLE_DEPTH_TEST);
}

// ============================================================
// PHYSICS UPDATE
// ============================================================
void updateFalling() {
  if (!falling) return;
  fallVelocity += gravity; currentSolid.centerY += fallVelocity;
  if (currentSolid.centerY + currentSolid.halfHeight() >= groundY) {
    currentSolid.centerY = groundY - currentSolid.halfHeight(); triggerImpact();
  }
}

void updateFallingAll() {
  if (!falling) return;
  fallVelocity += gravity; boolean hit = false;
  for (Solid s : allSolids) {
    s.centerY += fallVelocity;
    if (s.centerY + s.halfHeight() >= groundY) { s.centerY = groundY - s.halfHeight(); hit = true; }
  }
  if (hit) triggerImpact();
}

void updateJump() {
  if (jumpState == 3) return; // Landed

  jumpX += jumpVelX;
  if (frameCount % 2 == 0) jumpTrail.add(new PVector(jumpX, jumpY, 0));

  if (jumpState == 0) { // Flat approach
    jumpY = groundY - currentSolid.halfHeight();
    if (jumpX >= RAMP_START_X) jumpState = 1;
  } 
  else if (jumpState == 1) { // On Ramp
    float rampRatio = (jumpX - RAMP_START_X) / (RAMP_END_X - RAMP_START_X);
    jumpY = groundY - currentSolid.halfHeight() - (rampRatio * RAMP_HEIGHT);
    jumpPitch = atan2(RAMP_HEIGHT, RAMP_END_X - RAMP_START_X); // Pitch up
    
    if (jumpX >= RAMP_END_X) {
      jumpState = 2; 
      jumpVelY = -(RAMP_HEIGHT / (RAMP_END_X - RAMP_START_X)) * jumpVelX * 0.9; 
      jumpStartX = jumpX;
    }
  } 
  else if (jumpState == 2) { // Airborne
    jumpVelY += gravity; jumpY += jumpVelY;
    float currentH = groundY - currentSolid.halfHeight() - jumpY;
    if (currentH > jumpMaxHeight) jumpMaxHeight = currentH;
    
    // Dynamic pitch in air (slight nose dive)
    jumpPitch = constrain(atan2(-jumpVelY, jumpVelX), -0.4, 0.4);

    if (jumpY >= groundY - currentSolid.halfHeight()) {
      jumpY = groundY - currentSolid.halfHeight();
      jumpPitch = 0; jumpState = 3; jumpLength = jumpX - jumpStartX;
      triggerImpact();
    }
  }
}

void triggerImpact() {
  fallVelocity = 0.0; falling = false; impactActive = true; automaticImpact = true;
  impactProgress = 0.0; impactSlider.value = 0.0;
  currentStressMPa = peakStressMPa; currentReactionForce = peakReactionForce;
  currentImpulse = peakImpulse; impulseRatio = 1.0; impactStartTime = millis();
}

void updateImpact() {
  if (falling && currentScreen != SCREEN_JUMP) { currentStressMPa = 0; currentImpulse = 0; impulseRatio = 0; return; }
  if (currentScreen == SCREEN_JUMP && jumpState != 3) { currentStressMPa = 0; currentImpulse = 0; return; }
  if (!impactActive) return;

  if (automaticImpact) {
    impactProgress = constrain((millis() - impactStartTime) / IMPACT_DURATION_MS, 0, 1); 
    impactSlider.value = impactProgress;
  }

  float decay = exp(-DECAY_RATE * impactProgress);
  currentStressMPa = peakStressMPa * decay;
  currentImpulse = peakImpulse * decay; impulseRatio = decay;
  if (impactProgress >= 1.0) { impactActive = false; currentStressMPa = 0; currentImpulse = 0; impulseRatio = 0; }
}

// ENVIRONMENT RENDERING
// ============================================================
void drawGround(float w, float d) {
  pushMatrix(); translate(0, groundY + 5.0, 0); fill(175, 180, 185); stroke(90); strokeWeight(2); box(w, 10, w); popMatrix();
  pushMatrix(); translate(0, groundY, 0); stroke(145, 150, 155); strokeWeight(1);
  for (float x = -d; x <= d; x += 50) line(x, 0, -d, x, 0, d);
  for (float z = -d; z <= d; z += 50) line(-d, 0, z, d, 0, z);
  popMatrix();
}

void drawHeight() {
  if (!falling) return;
  float bottom = currentSolid.centerY + currentSolid.halfHeight();
  float h = groundY - bottom; if (h <= 1) return;
  float x = currentSolid.width() / 2.0 + 100;
  stroke(0, 110, 220); strokeWeight(3);
  line(x, bottom, 0, x, groundY, 0); line(x - 10, bottom, 0, x + 10, bottom, 0); line(x - 10, groundY, 0, x + 10, groundY, 0);
  drawWorldText("h = " + nf(h, 1, 1), x + 60, (bottom + groundY) / 2.0, 0, 22, color(0, 100, 210));
}

void drawFallingLoad(float top) {
  stroke(20); strokeWeight(8);
  line(0, top - 100, 0, 0, top - 10, 0); line(0, top - 10, 0, -16, top - 35, 0); line(0, top - 10, 0, 16, top - 35, 0);
  drawWorldText("P", 45, top - 85, 0, 30, color(20));
}

void drawImpulseVector(boolean showText) {
  if (impulseRatio <= 0.005) return;
  hint(DISABLE_DEPTH_TEST);
  float arrowLength = 200.0 * impulseRatio, endY = groundY - arrowLength;

  stroke(255, 40, 30, 45 + 80 * impulseRatio); strokeWeight(15 * impulseRatio + 3); line(0, groundY, 0, 0, endY, 0);
  stroke(225, 20, 20, 235); strokeWeight(7 * impulseRatio + 2); line(0, groundY, 0, 0, endY, 0);

  if (arrowLength > 12) {
    float ah = min(34, arrowLength * 0.28); 
    line(0, endY, 0, -ah, endY + ah, 0); line(0, endY, 0, ah, endY + ah, 0);
  }

  pushMatrix(); translate(0, groundY, 0); noStroke(); fill(255, 20, 20, 220); sphere(12 + 5 * impulseRatio); popMatrix();

  if (showText) {
    drawWorldText("IMPULSE J", 130, endY - 5, 0, 20, color(200, 0, 0));
    drawWorldText("J = " + nf(currentImpulse, 1, 0) + " N s", 125, endY + 23, 0, 16, color(150, 0, 0));
  }
  hint(ENABLE_DEPTH_TEST);
}

void drawWorldText(String s, float x, float y, float z, float size, color c) {
  pushMatrix(); translate(x, y, z); rotateY(-rotY); rotateX(-rotX); fill(c); textSize(size); text(s, 0, 0); popMatrix();
}

void drawSimulationHUD(String title) {
  hint(DISABLE_DEPTH_TEST); camera();
  fill(20); textSize(22); text(title, width / 2, 30);
  textSize(12); fill(75); text("Drag = Rotate | Pinch = Zoom", width / 2, 52);
  resetButton.draw(); menuButton.draw();

  if (falling) { fill(0, 110, 220); textSize(17); text("FREE FALL", width / 2, height - 285); } 
  else if (impactActive) { fill(210, 0, 0); textSize(18); text("COLLISION STRESS = " + nf(currentStressMPa, 1, 2) + " MPa", width / 2, height - 285); } 

  sliceSlider.draw("CROSS-SECTION SLICER"); impactSlider.draw("IMPACT TIME / STRESS PROPAGATION");

  fill(255, 255, 255, 238); stroke(100); strokeWeight(2); rect(10, height - 100, width - 20, 80, 10);
  fill(25); textSize(14); text("Stress(t) = 0.68 exp(-4t) MPa", width / 2, height - 70);
  fill(70); textSize(11); text(currentScreen == SCREEN_DROP_ALL ? "Comparing structures" : currentSolid.dimensionText(), width / 2, height - 40);
  hint(ENABLE_DEPTH_TEST);
}

// ============================================================
// CONTROLS & INPUT
// ============================================================
void mousePressed() { handlePress(mouseX, mouseY); }
void mouseDragged() {
  if (currentScreen != SCREEN_MENU) {
    if (impactSlider.handleDrag(mouseX, mouseY)) { automaticImpact = false; updateFromImpactSlider(); return; } 
    if (sliceSlider.handleDrag(mouseX, mouseY)) return; 
    rotateView(mouseX - previousTouchX, mouseY - previousTouchY);
  }
  previousTouchX = mouseX; previousTouchY = mouseY;
}
void mouseReleased() { sliceSlider.active = false; impactSlider.active = false; }

void touchStarted() {
  if (touches.length == 1) {
    float tx = touches[0].x, ty = touches[0].y; 
    if (currentScreen != SCREEN_MENU) { 
      if (impactSlider.contains(tx, ty)) { impactSlider.active = true; automaticImpact = false; impactSlider.value = constrain((tx - impactSlider.x) / impactSlider.w, 0, 1); updateFromImpactSlider(); return; } 
      if (sliceSlider.contains(tx, ty)) { sliceSlider.active = true; sliceSlider.value = constrain((tx - sliceSlider.x) / sliceSlider.w, 0, 1); return; } 
    } 
    previousTouchX = tx; previousTouchY = ty;
  }
  if (touches.length >= 2) { oldPinchDistance = dist(touches[0].x, touches[0].y, touches[1].x, touches[1].y); pinchActive = true; }
}

void touchMoved() {
  if (touches.length >= 2) {
    float newDist = dist(touches[0].x, touches[0].y, touches[1].x, touches[1].y); 
    cameraDistance = constrain(cameraDistance - (newDist - oldPinchDistance) * 2.0, minimumZoom, maximumZoom); 
    oldPinchDistance = newDist; pinchActive = true; return;
  }
  if (touches.length == 1) {
    float tx = touches[0].x, ty = touches[0].y; 
    if (impactSlider.active) { impactSlider.value = constrain((tx - impactSlider.x) / impactSlider.w, 0, 1); automaticImpact = false; updateFromImpactSlider(); return; } 
    if (sliceSlider.active) { sliceSlider.value = constrain((tx - sliceSlider.x) / sliceSlider.w, 0, 1); return; } 
    if (currentScreen != SCREEN_MENU) rotateView(tx - previousTouchX, ty - previousTouchY);
    previousTouchX = tx; previousTouchY = ty;
  }
}

void touchEnded() {
  if (!pinchActive && !sliceSlider.active && !impactSlider.active && touches.length == 0) handlePress(mouseX, mouseY);
  pinchActive = false; sliceSlider.active = false; impactSlider.active = false;
}

void handlePress(float x, float y) {
  if (currentScreen == SCREEN_MENU) {
    if (upButton.contains(x, y)) { selectedShape--; if (selectedShape < 0) selectedShape = SHAPE_COUNT - 1; createCurrentSolid(); } 
    else if (downButton.contains(x, y)) { selectedShape++; if (selectedShape >= SHAPE_COUNT) selectedShape = 0; createCurrentSolid(); } 
    else if (dropButton.contains(x, y)) { currentScreen = SCREEN_SIM; restartSimulation(); }
    else if (dropAllButton.contains(x, y)) { currentScreen = SCREEN_DROP_ALL; restartAllSimulation(); }
    else if (jumpButton.contains(x, y)) { currentScreen = SCREEN_JUMP; restartJump(); }
  } else {
    if (resetButton.contains(x, y)) { 
      if (currentScreen == SCREEN_SIM) restartSimulation(); 
      else if (currentScreen == SCREEN_DROP_ALL) restartAllSimulation(); 
      else restartJump(); 
    } 
    else if (menuButton.contains(x, y)) currentScreen = SCREEN_MENU;
  }
}

void rotateView(float dx, float dy) { rotY += dx * 0.01; rotX = constrain(rotX + dy * 0.01, -1.35, 1.35); }
void updateFromImpactSlider() { impactProgress = impactSlider.value; float decay = exp(-DECAY_RATE * impactProgress); currentStressMPa = peakStressMPa * decay; currentImpulse = peakImpulse * decay; impulseRatio = decay; impactActive = true; }

void restartSimulation() {
  falling = true; impactActive = false; automaticImpact = true; fallVelocity = 0.0; impactProgress = 0.0; impactSlider.value = 0.0; sliceSlider.value = 0.50; currentStressMPa = 0; currentImpulse = 0; impulseRatio = 0;
  if (currentSolid != null) currentSolid.centerY = groundY - currentSolid.halfHeight() - dropHeight;
}
void restartAllSimulation() { restartSimulation(); for (Solid s : allSolids) s.centerY = groundY - s.halfHeight() - dropHeight; }

void restartJump() {
  jumpX = -700; jumpY = groundY - currentSolid.halfHeight(); jumpVelX = 14.0; jumpVelY = 0; jumpState = 0; jumpMaxHeight = 0; jumpPitch = 0; jumpTrail.clear();
  impactActive = false; impactProgress = 0; impactSlider.value = 0; sliceSlider.value = 0.5; currentStressMPa = 0; impulseRatio = 0;
}

void setupCamera(float dist) {
  camera(sin(rotY) * cos(rotX) * dist, sin(rotX) * dist, cos(rotY) * cos(rotX) * dist, 0, 20, 0, 0, 1, 0);
  perspective(PI / 3.0, float(width) / float(height), 10.0, 5000.0);
}

color stressColor(float n, float strength) {
  n = constrain(n, 0, 1); color c;
  if (n < 0.25) c = lerpColor(color(255, 20, 20), color(255, 130, 0), map(n, 0, 0.25, 0, 1));
  else if (n < 0.50) c = lerpColor(color(255, 130, 0), color(255, 220, 0), map(n, 0.25, 0.50, 0, 1));
  else if (n < 0.75) c = lerpColor(color(255, 220, 0), color(35, 210, 70), map(n, 0.50, 0.75, 0, 1));
  else c = lerpColor(color(35, 210, 70), color(40, 100, 255), map(n, 0.75, 1.0, 0, 1));
  return color(red(c), green(c), blue(c), 175.0 * strength);
}

// ============================================================
// SOLID CLASSES (Primitives)
// ============================================================
abstract class Solid {
  String name; float centerY;
  Solid(String n) { name = n; }
  abstract float width(); abstract float depth(); abstract float totalHeight(); abstract float halfHeight(); abstract float contactArea();
  abstract void drawTransparentShell(); abstract void drawStress(float progress, float stress); abstract void drawCrossSection(float slider); abstract void drawAnnotations();

abstract class CompositeSolid extends Solid {
  float[][] parts; // {w, h, d, x, y, z, type(0=box, 1=X-Cylinder)}
  CompositeSolid(String n) { super(n); }
  
  void drawTransparentShell() {
    hint(DISABLE_DEPTH_MASK);
    pushMatrix(); translate(0, centerY, 0);
    for (float[] p : parts) {
      float type = p.length > 6 ? p[6] : 0;
      pushMatrix(); translate(p[3], p[4], p[5]);
      fill(100, 180, 225, 30); stroke(15, 50, 75, 200); strokeWeight(3);
      if (type == 0) box(p[0], p[1], p[2]);
      else if (type == 1) { rotateZ(HALF_PI); drawCylinderBand(p[1]/2.0, p[0], 0, color(100, 180, 225, 30), 24); }
      popMatrix();
    }
    popMatrix();
    hint(ENABLE_DEPTH_MASK);
  }

  void drawStress(float progress, float stress) {
    float strength = stress / peakStressMPa, front = lerp(totalHeight()*0.04, totalHeight(), progress);
    pushMatrix(); translate(0, centerY, 0);
    for (float[] p : parts) {
      float compW=p[0], compH=p[1], compD=p[2], compX=p[3], compY=p[4], compZ=p[5], type=p.length>6?p[6]:0;
      float compBot = compY + compH/2.0, solidBot = halfHeight();
      if (solidBot - compBot >= front) continue;

      int bands = max(2, int(compH / 5.0)); float bh = compH / bands;
      for (int i = 0; i < bands; i++) {
        float bY = compBot - i*bh - bh/2.0, fromSolBot = solidBot - bY;
        if (fromSolBot > front) continue;
        pushMatrix(); translate(compX, bY, compZ);
        fill(stressColor(fromSolBot / front, strength)); noStroke();
        if (type == 0) box(compW*0.94, bh+1.2, compD*0.94);
        else if (type == 1) {
          float dy = abs(bY - compY), val = (compH/2.0)*(compH/2.0) - dy*dy;
          if (val > 0) box(compW*0.94, bh+1.2, sqrt(val)*2.0 * 0.94); // Math cylinder chord
        }
        popMatrix();
      }
    }
    popMatrix();
  }

  void drawCrossSection(float slider) {
    float localY = map(slider, 0, 1, -halfHeight(), halfHeight());
    pushMatrix(); translate(0, centerY + localY, 0); rotateX(HALF_PI); fill(255,110,0,80); stroke(255, 110, 0); strokeWeight(5); rectMode(CENTER);
    for (float[] p : parts) {
      float type=p.length>6?p[6]:0, cT=p[4]-p[1]/2.0, cB=p[4]+p[1]/2.0;
      if (localY >= cT && localY <= cB) {
        pushMatrix(); translate(p[3], p[5], 0);
        if (type == 0) rect(0, 0, p[0]*0.96, p[2]*0.96);
        else if (type == 1) { float dy = abs(localY - p[4]), val = (p[1]/2.0)*(p[1]/2.0) - dy*dy; if(val>0) rect(0, 0, p[0]*0.96, sqrt(val)*2.0 * 0.96); }
        popMatrix();
      }
    }
    popMatrix();
  }
}

class UIButton {
  float x, y, w, h;
  String label;

  UIButton(float x, float y, float w, float h, String label) {
    this.x = x;
    this.y = y;
    this.w = w;
    this.h = h;
    this.label = label;
  }

  void draw() {
    hint(DISABLE_DEPTH_TEST);
    camera();
    
    fill(210, 215, 220);
    stroke(120);
    strokeWeight(2);
    rectMode(CORNER);
    rect(x, y, w, h, 6);
    
    fill(30);
    textAlign(CENTER, CENTER);
    textSize(h * 0.4);
    text(label, x + (w / 2.0), y + (h / 2.0));
    
    hint(ENABLE_DEPTH_TEST);
  }

  boolean contains(float px, float py) {
    return px >= x && px <= x + w && py >= y && py <= y + h;
  }
}

class UISlider {
  float x, y, w;
  float value;
  boolean active = false;

  UISlider(float x, float y, float w, float v) {
    this.x = x;
    this.y = y;
    this.w = w;
    this.value = v;
  }

  void draw(String title) {
    hint(DISABLE_DEPTH_TEST);
    camera();

    // Background container
    fill(255, 255, 255, 238);
    stroke(100);
    strokeWeight(2);
    rectMode(CORNER);
    rect(x - 10, y - 30, w + 20, 65, 8);

    // Title
    fill(0, 100, 200);
    textAlign(CENTER, TOP);
    textSize(14);
    text(title, x + w / 2, y - 20);

    // Track
    stroke(180);
    strokeWeight(6);
    line(x, y + 15, x + w, y + 15);

    // Active Track
    stroke(0, 120, 240);
    float knobX = x + (w * value);
    line(x, y + 15, knobX, y + 15);

    // Knob
    noStroke();
    fill(0, 100, 220);
    ellipse(knobX, y + 15, 24, 24);
    fill(255);
    ellipse(knobX, y + 15, 10, 10);

    hint(ENABLE_DEPTH_TEST);
  }

  boolean contains(float px, float py) {
    // Generous touch hit area
    return px >= x - 20 && px <= x + w + 20 && py >= y - 30 && py <= y + 40;
  }

  boolean handleDrag(float px, float py) {
    if (active) {
      value = constrain((px - x) / w, 0, 1);
      return true;
    }
    return false;
  }
}
